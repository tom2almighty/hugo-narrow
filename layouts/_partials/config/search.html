{{/* Search configuration resolver. `search: true` keeps the default on, `search: false` or `search.enabled: false` turns the feature off: the dock button, the dialog, the search bundle, and the config script are skipped, and `index.json` is emitted empty. */}} {{- $enabled := true -}} {{- $contentLength := 2000 -}} {{- $params := site.Params.search -}} {{- if reflect.IsMap $params -}} {{- if isset $params "enabled" -}} {{- $enabled = $params.enabled -}} {{- end -}} {{- if isset $params "indexcontentlength" -}} {{- $contentLength = $params.indexcontentlength -}} {{- end -}} {{- else if eq $params false -}} {{- $enabled = false -}} {{- end -}} {{- return dict "enabled" $enabled "indexContentLength" $contentLength "searchIndexURL" (printf "%sindex.json" site.Home.RelPermalink) -}}
