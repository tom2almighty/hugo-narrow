{{/* Gallery configuration resolver. Site params expose only `enabled` and `defaultLayout`; the remaining layout values are theme defaults and can still be overridden per page. Hugo lowercases every param key, so the merged values are read back by their lowercase name and re-emitted in the camelCase form SmartGallery expects. gallery.js can then pass them straight through without key fallbacks. */}} {{- $defaults := dict "defaultlayout" "justified" "gap" 10 "targetrowheight" 200 "lastrowbehavior" "center" "columnwidth" 200 "columns" 4 -}} {{- $globalGallery := site.Params.gallery | default dict -}} {{- $globalLightbox := site.Params.lightbox | default dict -}} {{- if not (reflect.IsMap $globalGallery) -}} {{- $globalGallery = dict "enabled" $globalGallery -}} {{- end -}} {{- if not (reflect.IsMap $globalLightbox) -}} {{- $globalLightbox = dict "enabled" $globalLightbox -}} {{- end -}} {{- $pageGallery := .Params.gallery | default dict -}} {{- $pageLightbox := .Params.lightbox | default dict -}} {{- if not (reflect.IsMap $pageGallery) -}} {{- $pageGallery = dict "enabled" $pageGallery -}} {{- end -}} {{- if not (reflect.IsMap $pageLightbox) -}} {{- $pageLightbox = dict "enabled" $pageLightbox -}} {{- end -}} {{- $galleryEnabled := cond (isset $pageGallery "enabled") $pageGallery.enabled ($globalGallery.enabled | default false) -}} {{- $lightboxEnabled := cond (isset $pageLightbox "enabled") $pageLightbox.enabled ($globalLightbox.enabled | default false) -}} {{- $pageHasImageFigure := .Store.Get "hasImageFigure" | default false -}} {{- $pageHasLightboxFigure := .Store.Get "hasLightboxFigure" | default false -}} {{- $galleryConfig := merge $defaults $globalGallery $pageGallery -}} {{- $galleryOptions := dict "defaultLayout" $galleryConfig.defaultlayout "gap" $galleryConfig.gap "targetRowHeight" $galleryConfig.targetrowheight "lastRowBehavior" $galleryConfig.lastrowbehavior "columnWidth" $galleryConfig.columnwidth "columns" $galleryConfig.columns -}} {{- return dict "gallery" (and $galleryEnabled $pageHasImageFigure) "lightbox" (and $lightboxEnabled (or $pageHasImageFigure $pageHasLightboxFigure)) "galleryOptions" $galleryOptions -}}
