{{/* 统一处理 JS 构建、压缩、指纹和加载方式。 参数： - path: 资源路径，与 resource 二选一 - resource: 已获取的资源，与 path 二选一 - targetPath: 输出路径，默认使用 path */}} {{- $resource := .resource -}} {{- $path := .path -}} {{- if and (not $resource) $path -}} {{- $resource = resources.Get $path -}} {{- end -}} {{- $targetPath := .targetPath | default $path -}} {{- with $resource -}} {{- $opts := dict "minify" (not hugo.IsDevelopment) "sourceMap" (cond hugo.IsDevelopment "external" "") "targetPath" $targetPath -}} {{- with . | js.Build $opts -}} {{- if hugo.IsDevelopment -}} {{- else -}} {{- with . | fingerprint -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
