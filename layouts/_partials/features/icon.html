{{/* Icon partial template Usage: {{ partial "features/icon.html" (dict "name" "palette" "class" "h-5 w-5" "id" "palette-icon") }} {{ partial "features/icon.html" (dict "name" "github" "size" "lg") }} Parameters: - name: Icon name (required) - corresponds to SVG file in assets/icons/ - class: CSS classes to apply to the SVG (optional) - id: HTML id attribute (optional) - size: Predefined size (optional) - "xs", "sm", "md", "lg", "xl", "2xl" - ariaLabel: Accessibility label (optional) - hidden: Whether to add aria-hidden="true" (optional, defaults to false) Icons are emitted as references to a fingerprinted site sprite, so repeated icons do not duplicate their path data in every page. */}} {{- $name := .name -}} {{- $class := .class | default "" -}} {{- $id := .id | default "" -}} {{- $size := .size | default "" -}} {{- $ariaLabel := .ariaLabel | default "" -}} {{- $hidden := .hidden | default false -}} {{/* Size presets */}} {{- $sizeClasses := dict "xs" "h-3 w-3" "sm" "h-4 w-4" "md" "h-5 w-5" "lg" "h-6 w-6" "xl" "h-8 w-8" "2xl" "h-10 w-10" -}} {{/* Build CSS classes */}} {{- $finalClass := $class -}} {{- if $size -}} {{- $sizeClass := index $sizeClasses $size -}} {{- if $sizeClass -}} {{- $finalClass = printf "%s %s" $sizeClass $class -}} {{- end -}} {{- end -}} {{/* Get the SVG file */}} {{- $iconPath := printf "icons/%s.svg" $name -}} {{- $icon := resources.Get $iconPath -}} {{- if $icon -}} {{- $sprite := partialCached "features/icon-sprite.html" . -}} {{/* Build attributes to add to SVG tag */}} {{- $attrs := "" -}} {{- if $finalClass -}} {{- $attrs = printf ` class="%s"` ($finalClass | strings.TrimSpace | htmlEscape) -}} {{- end -}} {{- if $id -}} {{- $attrs = printf `%s id="%s"` $attrs ($id | htmlEscape) -}} {{- end -}} {{- if $hidden -}} {{- $attrs = printf `%s aria-hidden="true"` $attrs -}} {{- else if $ariaLabel -}} {{- $attrs = printf `%s role="img" aria-label="%s"` $attrs ($ariaLabel | htmlEscape) -}} {{- else -}} {{- $attrs = printf `%s aria-hidden="true"` $attrs -}} {{- end -}} {{- $use := printf `<%s href="%s#icon-%s">` "use" $sprite.RelPermalink $name "use" -}} {{- printf `%s` $attrs $use | safeHTML -}} {{- else -}} {{/* Fallback if icon not found */}} {{/* Log warning in development */}} {{- if hugo.IsDevelopment -}} {{- warnf "Icon '%s' not found at %s" $name $iconPath -}} {{- end -}} {{- end -}}
