{{/* 生成站点级 SVG sprite，避免每个图标都内联重复的 path。 */}} {{- $symbols := slice -}} {{- range resources.Match "icons/*.svg" -}} {{- $name := path.Base .Name | strings.TrimSuffix ".svg" -}} {{- $content := .Content -}} {{- $content = replaceRE `(?s)]*)?>` (printf `` $name) $content 1 -}} {{- $content = replaceRE `` "" $content -}} {{- /* 只去掉根节点上的宽高，图标内部的 rect / line 尺寸必须保留。 */ -}} {{- $content = replaceRE `(]*?)\s+width="[^"]*"` "$1" $content -}} {{- $content = replaceRE `(]*?)\s+height="[^"]*"` "$1" $content -}} {{- $symbols = $symbols | append $content -}} {{- end -}} {{- $sprite := printf `` (delimit $symbols "") -}} {{- return (resources.FromString "icons/sprite.svg" $sprite | fingerprint) -}}
