{{/* 生成站点级 SVG sprite，避免每个图标都内联重复的 path。 */}} {{- $symbols := slice -}} {{- range resources.Match "icons/*.svg" -}} {{- $name := path.Base .Name | strings.TrimSuffix ".svg" -}} {{- $content := .Content -}} {{- $content = replaceRE `\s+width="[^"]*"` "" $content -}} {{- $content = replaceRE `\s+height="[^"]*"` "" $content -}} {{- $content = replaceRE `(?s)]*)?>` (printf `` $name) $content 1 -}} {{- $content = replaceRE `` "" $content -}} {{- $symbols = $symbols | append $content -}} {{- end -}} {{- $sprite := printf `` (delimit $symbols "") -}} {{- return (resources.FromString "icons/sprite.svg" $sprite | fingerprint) -}}
