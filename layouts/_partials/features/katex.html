{{/* KaTeX support. Delimiters are fixed; pages can only toggle the feature. */}} {{- $globalEnabled := false -}} {{- with site.Params.katex -}} {{- if eq (printf "%T" .) "bool" -}} {{- $globalEnabled = . -}} {{- else -}} {{- $globalEnabled = .enabled | default false -}} {{- end -}} {{- end -}} {{- $featureEnabled := $globalEnabled -}} {{- if isset .Params "katex" -}} {{- if eq (printf "%T" .Params.katex) "bool" -}} {{- $featureEnabled = .Params.katex -}} {{- else -}} {{- $featureEnabled = .Params.katex.enabled | default false -}} {{- end -}} {{- end -}} {{- $contentToScan := .Content -}} {{- $hasBlockMath := gt (len (findRE `\$\$` $contentToScan)) 0 -}} {{- $hasParenMath := gt (len (findRE `\\\(` $contentToScan)) 0 -}} {{- $hasBracketMath := gt (len (findRE `\\\[` $contentToScan)) 0 -}} {{- $hasInlineMath := gt (len (findRE `(^|[^$])\$[^$\n]+\$` $contentToScan)) 0 -}} {{- $shouldLoad := and $featureEnabled (or $hasBlockMath $hasParenMath $hasBracketMath $hasInlineMath) -}} {{- if $shouldLoad -}} {{- end -}}
