{{- /* Localized date formatting. The theme renders three variants: - long : post detail header, page headers - short : cards, archives, series - month : archive month heading (no Hugo token exists for "month year") Values accept either a Hugo localization token (":date_long", ":date_short") or a Go layout string ("2006年1月"). Per-language overrides go in languages..params.dateFormat. @context {time.Time} .date @context {string} .variant "long" | "short" | "month", default "long" */ -}} {{- $date := .date -}} {{- $variant := .variant | default "long" -}} {{- $defaults := dict "long" ":date_long" "short" ":date_short" "month" "January 2006" -}} {{- $configured := site.Params.dateFormat | default dict -}} {{- $format := index $configured $variant | default (index $defaults $variant) -}} {{- $date | time.Format $format -}}
