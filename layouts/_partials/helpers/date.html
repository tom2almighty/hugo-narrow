{{- /* Localized date formatting. The theme renders three variants: - long : post detail header, page headers - short : cards, archives, series - month : archive month heading (no Hugo token exists for "month year") Resolution order: 1. site.Params.dateFormat. (language params are already merged) 2. data/dateFormats.yaml, matched by locale, primary language, then language key 3. built-in Hugo localization tokens / English month-year fallback Values accept either a Hugo localization token (":date_long", ":date_short") or a Go layout string ("2006年1月"). @context {time.Time} .date @context {string} .variant "long" | "short" | "month", default "long" */ -}} {{- $date := .date -}} {{- $variant := .variant | default "long" -}} {{- $defaults := dict "long" ":date_long" "short" ":date_short" "month" "January 2006" -}} {{- $configured := site.Params.dateFormat | default dict -}} {{- $format := index $configured $variant -}} {{- if not $format -}} {{- $locale := lower site.Language.Locale -}} {{- $language := lower site.Language.Lang -}} {{- $dateFormats := hugo.Data.dateFormats | default dict -}} {{- $languageFormats := dict -}} {{- with index $dateFormats $locale -}}{{- $languageFormats = . -}}{{- end -}} {{- if not $languageFormats -}} {{- $primary := index (split $locale "-") 0 -}} {{- with index $dateFormats $primary -}}{{- $languageFormats = . -}}{{- end -}} {{- end -}} {{- if not $languageFormats -}} {{- with index $dateFormats $language -}}{{- $languageFormats = . -}}{{- end -}} {{- end -}} {{- with $languageFormats -}}{{- $format = index . $variant -}}{{- end -}} {{- end -}} {{- $format = $format | default (index $defaults $variant) -}} {{- $date | time.Format $format -}}
