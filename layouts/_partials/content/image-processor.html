{{/* 图片处理器 统一处理页面资源、全局资源和外部图片，并按需生成响应式 srcset。 参数： - context: 当前页面上下文 - src: 图片路径 - alt: 替代文本 - caption: 图片说明 - outputType: data | img | both - sizes: img 的 sizes 属性 - loading: img 的 loading 属性，默认 lazy - fetchpriority: img 的 fetchpriority 属性 */}} {{- $context := .context -}} {{- $src := .src | string -}} {{- $alt := .alt | default "" -}} {{- $caption := .caption | default "" -}} {{- $outputType := .outputType | default "both" -}} {{- $sizes := .sizes | default "" -}} {{- $loading := .loading | default "lazy" -}} {{- $fetchpriority := .fetchpriority | default "" -}} {{/* 初始化核心变量 */}} {{- $resource := false -}} {{- $resourceType := "" -}} {{- $isExternal := false -}} {{- $finalUrl := "" -}} {{- $searchPath := $src -}} {{- $isRootRelative := hasPrefix $src "/" -}} {{- $width := 0 -}} {{- $height := 0 -}} {{- $srcset := "" -}} {{/* 检查是否为外部URL */}} {{- if or (hasPrefix $src "http://") (hasPrefix $src "https://") (hasPrefix $src "data:") (hasPrefix $src "blob:") -}} {{- $isExternal = true -}} {{- $resourceType = "external" -}} {{- $finalUrl = $src -}} {{- else -}} {{/* 统一路径格式，避免带前导 / 的 relURL 在子路径站点下失效 */}} {{- if hasPrefix $src "./" -}} {{- $searchPath = strings.TrimPrefix "./" $src -}} {{- else if hasPrefix $src "/" -}} {{- $searchPath = strings.TrimPrefix "/" $src -}} {{- end -}} {{/* 根路径只查全局资源或静态文件；相对路径优先查页面资源 */}} {{- if not $isRootRelative -}} {{- with $context.Resources.GetMatch $searchPath -}} {{- $resource = . -}} {{- $resourceType = "page" -}} {{- $finalUrl = .RelPermalink -}} {{- end -}} {{- end -}} {{- if not $resource -}} {{- with resources.Get $searchPath -}} {{- $resource = . -}} {{- $resourceType = "global" -}} {{- $finalUrl = .RelPermalink -}} {{- else -}} {{- $resourceType = "static" -}} {{- $finalUrl = $searchPath | relURL -}} {{- end -}} {{- end -}} {{- end -}} {{/* 读取尺寸，并为静态资源可处理的格式生成 srcset */}} {{- if and $resource (not $isExternal) -}} {{- $mediaType := $resource.MediaType.SubType -}} {{- if in (slice "jpeg" "jpg" "png" "gif" "webp" "avif") $mediaType -}} {{- $width = $resource.Width -}} {{- $height = $resource.Height -}} {{- end -}} {{- if and (gt $width 0) (in (slice "jpeg" "jpg" "png" "webp" "avif") $mediaType) -}} {{- $parts := slice -}} {{- range (slice 480 768 1024 1440) -}} {{- if lt . $width -}} {{- $resized := $resource.Resize (printf "%dx" .) -}} {{- $parts = $parts | append (printf "%s %dw" $resized.RelPermalink .) -}} {{- end -}} {{- end -}} {{- if gt (len $parts) 0 -}} {{- $parts = $parts | append (printf "%s %dw" $finalUrl $width) -}} {{- $srcset = delimit $parts ", " -}} {{- end -}} {{- end -}} {{- end -}} {{/* 设置核心数据到上下文 */}} {{- if or (eq $outputType "data") (eq $outputType "both") -}} {{- $context.Store.Set "imageResource" $resource -}} {{- $context.Store.Set "imageUrl" $finalUrl -}} {{- $context.Store.Set "imageSrcset" $srcset -}} {{- $context.Store.Set "imageWidth" $width -}} {{- $context.Store.Set "imageHeight" $height -}} {{- $context.Store.Set "imageType" $resourceType -}} {{- $context.Store.Set "isExternal" $isExternal -}} {{- end -}} {{/* 生成图片HTML */}} {{- if or (eq $outputType "img") (eq $outputType "both") -}} {{- end -}}
