{{- /* Link card shortcode. Single card: {{< link title="Google" url="https://google.com" description="..." icon="..." >}} Group from data/links.yaml: {{< link group="social" >}} */ -}} {{- $title := .Get "title" -}} {{- $url := .Get "url" -}} {{- $description := .Get "description" -}} {{- $icon := .Get "icon" -}} {{- $group := .Get "group" -}} {{- if and $title $url -}}
{{ partial "content/link.html" (dict "title" $title "description" $description "url" $url "icon" $icon ) }}
{{- else if $group -}} {{- $links := where (hugo.Data.links | default slice) "group" $group -}} {{- if not $links -}} {{- errorf "link shortcode: no entries for group %q in data/links.yaml" $group -}} {{- end -}}
{{- range $index, $link := $links -}} {{- if not (and $link.title $link.url) -}} {{- errorf "link shortcode: entry %d in group %q needs title and url" $index $group -}} {{- end -}} {{ partial "content/link.html" $link }} {{- end -}}
{{- else -}} {{- errorf "link shortcode: pass title + url for one card, or group to render data/links.yaml" -}} {{- end -}}
